PUSH 0     ; sum
PUSH 0     ; i
PUSH 0     ; j
CALL nested 3
HALT

nested:
; local 0 → sum
; local 1 → i
; local 2 → j

PUSH 0
STORE_LOCAL 0     ; sum = 0

PUSH 1
STORE_LOCAL 1     ; i = 1

outer_loop:
LOAD_LOCAL 1
PUSH 4
LT
JZ end_outer

PUSH 1
STORE_LOCAL 2     ; j = 1

inner_loop:
LOAD_LOCAL 2
PUSH 4
LT
JZ end_inner

LOAD_LOCAL 0
LOAD_LOCAL 1
LOAD_LOCAL 2
MUL
ADD
STORE_LOCAL 0

LOAD_LOCAL 2
PUSH 1
ADD
STORE_LOCAL 2
JMP inner_loop

end_inner:
LOAD_LOCAL 1
PUSH 1
ADD
STORE_LOCAL 1
JMP outer_loop

end_outer:
LOAD_LOCAL 0
RET
