PUSH 6        ; initial argument

CALL fact 1
HALT

fact:
LOAD_LOCAL 0
DUP           ; [n, n]
JZ base       ; if n == 0 → base case (consumes one n)

DUP           ; [n, n]
PUSH 1
SUB           ; [n, n-1]

CALL fact 1    ; [n, fact(n-1)]
MUL           ; [n * fact(n-1)]
RET

base:
DROP          ; remove the 0
PUSH 1        ; factorial(0) = 1
RET
