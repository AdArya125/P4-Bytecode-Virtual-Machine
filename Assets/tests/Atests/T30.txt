; ---------- T30: 5 Nested Loops ----------

PUSH 0   ; sum
PUSH 1   ; i1
PUSH 0   ; i2
PUSH 0   ; i3
PUSH 0   ; i4
PUSH 0   ; i5
CALL nested5 6
HALT

nested5:

L1:
LOAD_LOCAL 1
PUSH 3
LT
JZ END

PUSH 1
STORE_LOCAL 2
L2:
LOAD_LOCAL 2
PUSH 3
LT
JZ INC1

PUSH 1
STORE_LOCAL 3
L3:
LOAD_LOCAL 3
PUSH 3
LT
JZ INC2

PUSH 1
STORE_LOCAL 4
L4:
LOAD_LOCAL 4
PUSH 3
LT
JZ INC3

PUSH 1
STORE_LOCAL 5
L5:
LOAD_LOCAL 5
PUSH 3
LT
JZ INC4

; sum++
LOAD_LOCAL 0
PUSH 1
ADD
STORE_LOCAL 0

LOAD_LOCAL 5
PUSH 1
ADD
STORE_LOCAL 5
JMP L5

INC4:
LOAD_LOCAL 4
PUSH 1
ADD
STORE_LOCAL 4
JMP L4

INC3:
LOAD_LOCAL 3
PUSH 1
ADD
STORE_LOCAL 3
JMP L3

INC2:
LOAD_LOCAL 2
PUSH 1
ADD
STORE_LOCAL 2
JMP L2

INC1:
LOAD_LOCAL 1
PUSH 1
ADD
STORE_LOCAL 1
JMP L1

END:
LOAD_LOCAL 0
RET
